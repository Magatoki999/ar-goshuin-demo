  { name: '令和', start: new Date(2019, 4, 1) },

];

元号年を計算し、1年目のみ「元」と表示する（慣例：令和元年）。2年目以降は漢数字テーブルで変換する。

  const y = marginTop + fontSize + i * lineHeight;

  ctx.shadowColor = 'transparent';
  ctx.strokeStyle = 'rgba(0, 0, 0, 0.9)';
  ctx.lineWidth   = 5;
  ctx.lineJoin    = 'round';
  ctx.strokeText(ch, x, y);
  // ② 本体（金色）
  ctx.fillStyle   = '#D4AF37';
  ctx.shadowColor = 'rgba(0, 0, 0, 0.5)';
  ctx.shadowBlur  = 4;
  ctx.fillText(ch, x, y);
});
一文字ずつ2段階で描画することで視認性を確保する。まず strokeText で黒の縁取り（線幅5・角丸）を描き、その上から fillText で金色（#D4AF37）を重ねる。縁取りを先に描くことで文字の輪郭が背景の墨色に溶け込まず、どんな背景でも浮き上がって見える。lineJoin: 'round' で角が尖らず筆文字らしい丸みが出る。

アクセスログ（Googleフォーム）

GitHubPages はサーバーサイド処理ができないため、CSVへの直接書き込みは不可。Googleフォームへのfetch POSTを使うことで、認証画面を一切出さずにスプレッドシートへ記録できる。 
const FORM_ACTION = 'https://docs.google.com/forms/d/e/xxxx/formResponse';
const FORM_FIELDS = {
  timestamp:  'entry.209041019',
  lang:       'entry.1632396277',
  weather:    'entry.134011435',
  timeOfDay:  'entry.58239131',
  event:      'entry.957978847',
};
FORM_ACTION はGoogleフォームの公開URL末尾を viewform → formResponse に変えたもの。FORM_FIELDS の各 entry.XXXXXXXXX はフォームの各質問に対応するフィールドID。フォームのプレビューページでDevToolsの Console に document.querySelectorAll('[name^="entry."]').forEach(el => console.log(el.name)) を実行すると取得できる。
function sendLog(eventName) {
  const body = new FormData();
  body.append(FORM_FIELDS.timestamp, new Date().toISOString());
  body.append(FORM_FIELDS.lang,      LANG);
  body.append(FORM_FIELDS.weather,   currentWeather);
  body.append(FORM_FIELDS.timeOfDay, timeOfDay);
  body.append(FORM_FIELDS.event,     eventName);
  fetch(FORM_ACTION, { method: 'POST', body, mode: 'no-cors' }).catch(() => {});
}
FormData に各フィールドを追加して POST 送信。mode: 'no-cors' を指定するため fetch はレスポンスを読めないが、Googleフォームへの送信自体は正常に届く。.catch(() => {}) でネットワークエラーを握りつぶしてアプリを止めない。

// targetFound 内

  sendLog('ar_found');
  arLogSent = true;
}
arLogSent フラグで、ターゲットを見失って再認識するたびに記録されるのを防ぐ。1セッション（ページリロードまで）で ar_found は1回だけ送信される。

accessDOMContentLoaded（ページ到達時）リロードごとに1回
ar_foundARターゲット認識時arLogSentフラグで1セッション1回
capture撮影ボタンタップ時制限なし（複数回撮影可）
